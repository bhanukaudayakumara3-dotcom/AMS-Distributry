            const selector = document.getElementById('branchSelector');
            selector.innerHTML = `

            currentBranchKey = document.getElementById('branchSelector').value;
            renderAll();
        }

            let activeBranch = branchesData[currentBranchKey];
            document.getElementById('activeBranchTag').innerText = activeBranch.name;

            document.getElementById('settingBranchLoc').value = activeBranch.location;

            renderShopItems();
            renderAdminItems();
            renderCustomers();
            renderStaffSalary();
        }

            ['shopSection', 'adminItemsSection', 'customersSection', 'salarySection', 'settingsSection'].forEach(s => {
                document.getElementById(s).classList.add('hidden');
            });
            document.getElementById(id).classList.remove('hidden');
        }

        function saveItem(e) {
            e.preventDefault();
            let idx = parseInt(document.getElementById('editItemIndex').value);
            let itemObj = {
                name: document.getElementById('itemName').value,
                price: parseFloat(document.getElementById('itemPrice').value),
                image: document.getElementById('itemImage').value,
                desc: document.getElementById('itemDesc').value
            };

            if (idx === -1) {
                branchesData[currentBranchKey].items.push(itemObj);
            } else {
                branchesData[currentBranchKey].items[idx] = itemObj;
            }

            resetItemForm();
            renderAll();
            alert("භාණ්ඩය සාර්ථකව Save විය!");
        }

        function editItem(index) {
            let item = branchesData[currentBranchKey].items[index];
            document.getElementById('editItemIndex').value = index;
            document.getElementById('itemName').value = item.name;
            document.getElementById('itemPrice').value = item.price;
            document.getElementById('itemImage').value = item.image;
            document.getElementById('itemDesc').value = item.desc;
        }

        function deleteItem(index) {
            if (confirm("මෙම භාණ්ඩය ඉවත් කිරීමට අවශ්‍යද?")) {
                branchesData[currentBranchKey].items.splice(index, 1);
                renderAll();
            }
        }

        function resetItemForm() {
            document.getElementById('itemForm').reset();
            document.getElementById('editItemIndex').value = "-1";
        }

        // --- BRANCH SETTINGS LOGIC ---

            e.preventDefault();

            branchesData[currentBranchKey].location = document.getElementById('settingBranchLoc').value;

        }

            let staff = branchesData[currentBranchKey].staff[index];
            let newName = prompt("සේවකයාගේ නම ඇතුළත් කරන්න:", staff.name);
            let newRole = prompt("තනතුර ඇතුළත් කරන්න:", staff.role);
            let newBasic = prompt("මූලික වැටුප (Basic Salary):", staff.basic);
            let newAllow = prompt("දීමනාව (Allowance):", staff.allowance);

            if (newName !== null && newRole !== null && newBasic !== null && newAllow !== null) {
                staff.name = newName.trim() || staff.name;
                staff.role = newRole.trim() || staff.role;
                staff.basic = parseFloat(newBasic) || staff.basic;
                staff.allowance = parseFloat(newAllow) || staff.allowance;
                renderStaffSalary();
                alert("සේවක විස්තර සාර්ථකව Update විය!");
            }
        }

        // --- CUSTOMER & ORDER LOGIC ---
        function openOrderModal(itemName, itemPrice) {
            document.getElementById('orderItemName').value = itemName;
            document.getElementById('custDue').value = itemPrice;
            document.getElementById('orderModal').classList.remove('hidden');
        }

        function closeOrderModal() {
            document.getElementById('orderModal').classList.add('hidden');
        }

        function submitOrder(e) {
            e.preventDefault();
            let cust = {
                name: document.getElementById('custName').value,
                address: document.getElementById('custAddress').value,
                nic: document.getElementById('custNic').value,
                phone: document.getElementById('custPhone').value,
                item: document.getElementById('orderItemName').value,
                due: parseFloat(document.getElementById('custDue').value)
            };
            branchesData[currentBranchKey].customers.push(cust);
            closeOrderModal();
            document.getElementById('orderModal').querySelector('form').reset();
            renderCustomers();
            showSection('customersSection');
            alert("ගාහකයා සාර්ථකව එකතු විය!");
        }

        function renderCustomers() {
            const tbody = document.getElementById('customersTableBody');
            tbody.innerHTML = '';
            branchesData[currentBranchKey].customers.forEach((c, idx) => {
                tbody.innerHTML += `
                    <tr>
                        <td class="p-4 font-bold">${c.name}</td>
                        <td class="p-4">${c.address}</td>
                        <td class="p-4">${c.nic}</td>
                        <td class="p-4">${c.phone}</td>
                        <td class="p-4 font-semibold text-indigo-700">${c.item}</td>
                        <td class="p-4 font-extrabold text-red-600">LKR ${c.due.toLocaleString()}</td>
                        <td class="p-4 space-x-1">
                            <button onclick="payInstallment(${idx})" class="px-2.5 py-1 bg-emerald-600 text-white rounded text-xs font-semibold">ගෙවීම්</button>
                            <button onclick="deleteCustomer(${idx})" class="px-2.5 py-1 bg-red-600 text-white rounded text-xs font-semibold">Clear</button>
                        </td>
                    </tr>
                `;
            });
        }

        function payInstallment(index) {
            let paid = prompt("ගෙවූ මුදල ඇතුළත් කරන්න:");
            if (paid && !isNaN(paid)) {
                let cust = branchesData[currentBranchKey].customers[index];
                cust.due -= parseFloat(paid);
                if (cust.due <= 0) {
                    alert("සම්පූර්ණ මුදල ගෙවා අවසන්ය! ගාහකයා ඉවත් කෙරේ.");
                    branchesData[currentBranchKey].customers.splice(index, 1);
                } else {
                    alert("ඉතිරි මුදල යාවත්කාලීන විය.");
                }
                renderCustomers();
            }
        }

        function deleteCustomer(index) {
            if (confirm("මෙම ගාහකයා ඉවත් කිරීමට අවශ්‍යද?")) {
                branchesData[currentBranchKey].customers.splice(index, 1);
                renderCustomers();
            }
        }

        // Initialize on load
        initApp();
